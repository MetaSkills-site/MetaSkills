# Blueprint: AI-Assisted Bulk Question Import - Implementation Spec

## 1. Backend: AI Parsing Endpoint
**File:** `metaskills-backend/controllers/genai.controller.js`
**Endpoint:** `POST /api/genai/parse-questions`

**Request Body:**
```json
{
  "rawText": "String content extracted from PDF..."
}
```

**Logic / Pseudo-code:**
```javascript
const parseQuestionsText = async (req, res) => {
  try {
    const { rawText } = req.body;
    const prompt = `
      Extract multiple-choice questions from the following text. 
      Return ONLY a valid JSON array of objects. 
      Each object must match this schema:
      {
        "questionText": "string",
        "options": ["string", "string"],
        "correctOptionIndex": number // 0-based index
      }
      Text:
      ${rawText}
    `;
    
    // Call Gemini API
    const aiResponse = await geminiService.generateContent(prompt);
    
    // Clean markdown formatting if AI wraps in ```json
    let jsonString = aiResponse.replace(/```json/g, '').replace(/```/g, '').trim();
    const drafts = JSON.parse(jsonString);
    
    res.status(200).json({ success: true, drafts });
  } catch (error) {
    res.status(500).json({ success: false, message: "AI Parsing Failed", error: error.message });
  }
};
```

## 2. Backend: Bulk Import Endpoint
**File:** `metaskills-backend/controllers/quiz.controller.js`
**Endpoint:** `POST /api/quiz/bulk-import`

**Request Body:**
```json
{
  "moduleId": "60d5ec49f1b2c8a14c8b4567",
  "questions": [
    {
      "_id": "60d5ec49f1b2c8a14c8b4568", // Generated by frontend
      "content": [{ "type": "paragraph", "children": [{ "text": "What is Node?" }] }], // Slate.js format
      "options": [...],
      "correctOption": "ObjectId...",
      "explanation": [...]
    }
  ]
}
```

**Logic / Pseudo-code:**
```javascript
const bulkImportQuestions = async (req, res) => {
  const session = await mongoose.startSession();
  session.startTransaction();
  
  try {
    const { moduleId, questions } = req.body;
    
    // 1. Strict Validation
    if (!moduleId || !Array.isArray(questions) || questions.length === 0) {
      throw new Error("Invalid payload");
    }
    
    // 2. Insert Questions
    const insertedQuestions = await Questions.insertMany(questions, { session });
    
    // 3. Extract IDs
    const questionIds = insertedQuestions.map(q => q._id);
    
    // 4. Update Module
    const updatedModule = await Module.findByIdAndUpdate(
      moduleId,
      { $push: { questionIds: { $each: questionIds } } },
      { session, new: true }
    );
    
    if (!updatedModule) throw new Error("Module not found");
    
    await session.commitTransaction();
    session.endSession();
    
    res.status(201).json({ success: true, message: "Import successful", data: updatedModule });
  } catch (error) {
    await session.abortTransaction();
    session.endSession();
    res.status(400).json({ success: false, error: error.message });
  }
};
```
